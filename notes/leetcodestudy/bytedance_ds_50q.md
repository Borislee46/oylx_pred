### SQL 核心模板（必背）

```sql
-- 1. 连续登录 (login_date - ROW_NUMBER) ← P0 必须刻进肌肉记忆
WITH t1 AS (
    SELECT user_id, login_date,
           ROW_NUMBER() OVER (PARTITION BY user_id ORDER BY login_date) AS rn
    FROM (SELECT DISTINCT user_id, login_date FROM login_log) t
)
SELECT user_id, DATE_SUB(login_date, INTERVAL rn DAY) AS grp, COUNT(*) AS streak
FROM t1 GROUP BY user_id, grp;

-- 2. 留存率 (LEFT JOIN + DATE_ADD) ← P0
SELECT f.first_date,
       COUNT(DISTINCT f.user_id) AS total,
       COUNT(DISTINCT l.user_id) AS retained
FROM first_login f
LEFT JOIN login_log l ON f.user_id = l.user_id
    AND l.login_date = DATE_ADD(f.first_date, INTERVAL 1 DAY)
GROUP BY f.first_date;

-- 3. 漏斗 (GROUP BY + CASE WHEN) ← P0
SELECT user_id,
       MAX(CASE WHEN page='首页' THEN 1 ELSE 0 END) AS s1,
       MAX(CASE WHEN page='搜索' THEN 1 ELSE 0 END) AS s2
FROM visits GROUP BY user_id;

-- 4. TopN (ROW_NUMBER + WHERE rn = N) ← P1
SELECT * FROM (
    SELECT *, ROW_NUMBER() OVER (PARTITION BY group_col ORDER BY val DESC) AS rn
    FROM t
) tmp WHERE rn = 1;
```
